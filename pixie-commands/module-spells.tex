\def\pixieSpellSectionMinWidth{420}
\newcommand{\pixieLayoutSpellSection}[4]{
    % LAYOUT RESERVATION
    \pscustom{\translate(#1,#2)\pixieOutlinedSectionSimpleBackground{#3}{#4}}
    \rput(\fpeval{#1+(#3/2)},\fpeval{#2+(#4/2)}){\pixieFontGermanCaps Spell List}
}

\newcommand{\pixiePlaceSpellSection}[4]{
    \pixieOutlinedSectionSpell{#1}{#2}{#3}{#4}
    \psline[fillstyle=none,linestyle=solid,linecolor=gray!10!white,linewidth=0.5](#1,\fpeval{#2+#4-95+4.65})(\fpeval{#1+#3},\fpeval{#2+#4-95+4.65})
    \pixieOutlinedSectionSpellDividers{#1}{#2}{#3}{\fpeval{#4-95}}
    \pixieMakeSpellSlotsA{#1}{\fpeval{#4-78}}{#3} % width = 420 (initially 360), height = 100
    %\pixieMakeSpellSlotsB{625}{740} % width = 135, height = 220

    \pixieMakeSpellList{#1}{\fpeval{#2+15}}{#3}{\fpeval{#4-110}}
    \pixieStyleDefault
}
