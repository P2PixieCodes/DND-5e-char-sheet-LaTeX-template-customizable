\newcommand{\pixiePlaceInspiration}[2]{ % (#1,#2) = coordinates
    % default for letter = (124.4251,921.5568)
    % default for a4 = (119.4251,921.5568) 
    \pixieStyleStatsOutlines\pscustom{\translate(#1,#2)\pixieGraphicInspiration}
    \rput(\fpeval{#1+93.7812156},\fpeval{#2+18.75931158}){\pixieTextBold\pixieColorShortcutOne{\MakeUppercase{\pixieTermInspiration}}}

    \pixieStyleDefault
}
