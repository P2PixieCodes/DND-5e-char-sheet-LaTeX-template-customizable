\newcommand{\pixieMakeSectionGrid}[5]{
    % why `\psclip`? bc i might mess up the math ahaha
    \psclip{\pscustom[linestyle=none,fillstyle=none]{\translate(#1,#2) #5}}
        \pixieStyleSectionGrid
        \multips(\fpeval{#1+7},\fpeval{#2+10})(0,6){\fpeval{floor((#4-14)/6)}}{\psline(0,0)(\fpeval{#3-14},0)}
        \multips(\fpeval{#1+10},\fpeval{#2+5})(6,0){\fpeval{floor((#3-13)/6)}}{\psline(0,0)(0,\fpeval{#4-10})}
    \endpsclip
}
% regex search: \\psclip\{\\pscustom\[linestyle=none,fillstyle=none\]\{\\translate\(#1,#2\)\\pixie([a-zA-Z]*)\{#3\}\{#4\}\}\}\s*\n\s*\\pixieStyleSectionGrid\n\s*\\multips\([^\n]*\}\n\s*\\multips\([^\n]*\}\n\s*\\endpsclip
% regex replace: \\pixieMakeSectionGrid{#1}{#2}{#3}{#4}{\\pixie$1{#3}{#4}}

\newcommand{\pixieMakeSectionGridReverseNonClip}[4]{
    \pixieStyleSectionGrid
    \multips(\fpeval{#1+5},\fpeval{#2+#4-4})(0,-6){\fpeval{floor((#4-2)/6)}}{\psline(0,0)(\fpeval{#3-10},0)}
    \multips(\fpeval{#1+10},\fpeval{#2+#4})(6,0){\fpeval{floor((#3-10)/6)}}{\psline(0,0)(0,-\fpeval{#4-10})}
}
