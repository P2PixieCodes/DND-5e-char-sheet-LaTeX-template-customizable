% MAIN COMMAND
\newcommand{\pixiePlaceSpeciesSection}[4]{ % (#1,#2) = coords; #3 = width; #4 = height
    \pixieStyleSectionBackground\pscustom{\translate(#1,#2)\pixieSelectedSpeciesSectionBackground{#3}{#4}}
    \pixieMakeSectionGrid{#1}{#2}{#3}{#4}{\pixieSelectedSpeciesSectionBackground{#3}{#4}}
    \pixieStyleSectionOutline\pscustom{\translate(#1,#2)\pixieSelectedSpeciesSectionOutline{#3}{#4}}
    \pixieMakeLabelBox{#1}{#2}{#3}{\pixieLabelLengthSpecies}{\pixieSectionLabelTextSpecies}
    
    \iftoggle{interactive}{
        \put(\fpeval{#1+7},\fpeval{#2+13.5}){\pixieTextField[fontsize=6pt,height=\fpeval{(#4-16)*0.75}pt,width=\fpeval{(#3-14)*0.75}pt,setFf=Multiline]{speciesFeatures}}
    }{}
    \pixieStyleDefault
}
