% MAIN COMMAND
\newcommand{\pixieMakeSpellList}[4]{
    \pixieStyleWeaponTable
    % VERTICAL LINES
    \pscustom{
        % Grad
        \moveto(\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA},\fpeval{#2+\pixieSpellRowOffset})\rlineto(0,\fpeval{#4-18})
        % Name
        \moveto(\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}},\fpeval{#2+\pixieSpellRowOffset})\rlineto(0,\fpeval{#4-18})
        % Zeitaufwand
        \moveto(\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC},\fpeval{#2+\pixieSpellRowOffset})\rlineto(0,\fpeval{#4-18})
        % Reichweite
        \moveto(\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD},\fpeval{#2+\pixieSpellRowOffset})\rlineto(0,\fpeval{#4-18})
        % Konzentration, Ritual & erf. Materialien
        \moveto(\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+\pixieSpellColE},\fpeval{#2+\pixieSpellRowOffset})\rlineto(0,\fpeval{#4-18})
        % Notizen
    }
    % COLUMN TITLES
    {\pixieTextRegular\smallerInGerman\color{gray!60!black}
        \pixieRput[lB]
            {\fpeval{#1+\pixieSpellColOffset+2.5}}
            {\fpeval{#2+#4-20}}
            {\myuline{\pixieTermLevelspell}}
        \pixieRput[lB]
			{\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+2.5}}
			{\fpeval{#2+#4-20}}
            {\myuline{\pixieTermName}}
        \pixieRput[lb]
			{\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+2.5}}
			{\fpeval{#2+#4-20-2.75}}
            {\pixieTermCastingTimeSplitUnderline}
        \pixieRput[lB]
			{\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+2.5}}
			{\fpeval{#2+#4-20}}
            {\myuline{\pixieTermRange}}
        \pixieRput[lb]
            {\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2.5}}
            {\fpeval{#2+#4-20-2.75}}
            {\pixieTermConcentrationRitualRequiredMaterialsSplitUnderline}
        \pixieRput[lB]
            {\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+\pixieSpellColE+2.5}}
            {\fpeval{#2+#4-20}}
            {\myuline{\pixieTermNotes}}
    }
    % HORIZONTAL LINES, FIELDS, CHECKMARKS
    \multido{\iA=0+1}{\fpeval{floor(#4/24)-1}}{
        % Grad
        \psline
            (\fpeval{#1+\pixieSpellColOffset},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA-2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
        % Name
        \psline
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}-2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
        % Zeitaufwand
        \psline
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC-2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
        % Reichweite
        \psline
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD-2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
        % Konzentration, Ritual & erf. Materialien
        %\psline(\fpeval{25+85+40+48+2},0)(\fpeval{25+85+40+48+80-2},0)
        \pixieGraphicCheckStarLine
            {\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2}}
            {\fpeval{#2+\pixieSpellRowOffset+5+\pixieSpellRowHeight*\iA}}
        \pixieRput[l]
			{\fpeval{22+#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2}}
			{\fpeval{#2+\pixieSpellRowOffset+5+6+\pixieSpellRowHeight*\iA}}
			{\pixieTextRegular\pixieColorShortcutThree{\pixieTermConcentrationInitial}}
        \pixieGraphicCheckStarLine
            {\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2+\pixieSpellColEa}}
            {\fpeval{#2+\pixieSpellRowOffset+5+\pixieSpellRowHeight*\iA}}
        \pixieRput[l]
			{\fpeval{22+#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2+\pixieSpellColEa}}
			{\fpeval{#2+\pixieSpellRowOffset+5+6+\pixieSpellRowHeight*\iA}}
			{\pixieTextRegular\pixieColorShortcutThree{\pixieTermRitualInitial}}
        \pixieGraphicCheckStarLine
            {\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2+\pixieSpellColEb}}
            {\fpeval{#2+\pixieSpellRowOffset+5+\pixieSpellRowHeight*\iA}}
        \pixieRput[l]
			{\fpeval{22+#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+2+\pixieSpellColEb}}
			{\fpeval{#2+\pixieSpellRowOffset+5+6+\pixieSpellRowHeight*\iA}}
			{\pixieTextRegular\pixieColorShortcutThree{\pixieTermMaterialInitial}}
        % Notizen
        \psline
            (\fpeval{#1+\pixieSpellColOffset+\pixieSpellColA+\pixieSpellColB{#3}+\pixieSpellColC+\pixieSpellColD+\pixieSpellColE+2},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
            (\fpeval{#1+#3-10},\fpeval{#2+\pixieSpellRowOffset+\pixieSpellRowHeight*\iA})
    }
    % SECTION LABEL
    \pixieMakeLabelBox{#1}{\fpeval{#2-2}}{#3}{\pixieLabelLengthSpellList}{\pixieSectionLabelTextSpellList}
    % FILLABLE FIELDS
    \iftoggle{interactive}{
        \pixieMakeSpellEntryFillable{#1}{#2}{#3}{0}{Zero}
        \pixieMakeSpellEntryFillable{#1}{#2}{#3}{1}{One}
